CREATE VIEW resolved_metadata_conflicts AS
SELECT
    mm.sample_id,
    mm.field,
    mm.value,
    mm.source_paper,
    mm.source_doi,
    mm.source_table,
    mm.review_status,
    mm.id AS manual_metadata_id,
    mm.updated_at
FROM manual_metadata mm
WHERE mm.review_status = 'REVIEWED'
  AND EXISTS (
    SELECT 1 FROM manual_metadata mm2
    WHERE mm2.sample_id = mm.sample_id
      AND mm2.field = mm.field
      AND mm2.review_status = 'REVIEWED'
      AND mm2.value != mm.value
  )
ORDER BY mm.sample_id, mm.field, mm.updated_at DESC

CREATE VIEW resolved_metadata AS
WITH manual_reviewed AS (
    SELECT sample_id, field,
           value,
           id as manual_id,
           source_type,
           source_doi,
           ROW_NUMBER() OVER (PARTITION BY sample_id, field ORDER BY updated_at DESC) as rn,
           COUNT(*) OVER (PARTITION BY sample_id, field) as n_reviewed
    FROM manual_metadata
    WHERE review_status = 'REVIEWED'
),
manual_unique AS (
    SELECT sample_id, field, value, manual_id, source_type, source_doi, n_reviewed
    FROM manual_reviewed WHERE rn = 1
),
manual_conflict AS (
    SELECT sample_id, field
    FROM manual_reviewed
    GROUP BY sample_id, field
    HAVING COUNT(DISTINCT value) > 1
),
manual_pivot AS (
    SELECT
        mu.sample_id,
        MAX(CASE WHEN mu.field = 'host' AND mc.sample_id IS NULL THEN mu.value END) AS host,
        MAX(CASE WHEN mu.field = 'host' AND mc.sample_id IS NULL THEN 'manual' END) AS host_source,
        MAX(CASE WHEN mu.field = 'host' AND mc.sample_id IS NULL THEN mu.manual_id END) AS host_manual_id,
        MAX(CASE WHEN mu.field = 'organism_name' AND mc.sample_id IS NULL THEN mu.value END) AS organism_name,
        MAX(CASE WHEN mu.field = 'cultivar' AND mc.sample_id IS NULL THEN mu.value END) AS cultivar,
        MAX(CASE WHEN mu.field = 'cultivar' AND mc.sample_id IS NULL THEN 'manual' END) AS cultivar_source,
        MAX(CASE WHEN mu.field = 'isolation_source' AND mc.sample_id IS NULL THEN mu.value END) AS isolation_source,
        MAX(CASE WHEN mu.field = 'isolation_source' AND mc.sample_id IS NULL THEN 'manual' END) AS isolation_source_source,
        MAX(CASE WHEN mu.field = 'country' AND mc.sample_id IS NULL THEN mu.value END) AS country,
        MAX(CASE WHEN mu.field = 'country' AND mc.sample_id IS NULL THEN 'manual' END) AS country_source,
        MAX(CASE WHEN mu.field = 'region' AND mc.sample_id IS NULL THEN mu.value END) AS region,
        MAX(CASE WHEN mu.field = 'region' AND mc.sample_id IS NULL THEN 'manual' END) AS region_source,
        MAX(CASE WHEN mu.field = 'locality' AND mc.sample_id IS NULL THEN mu.value END) AS locality,
        MAX(CASE WHEN mu.field = 'locality' AND mc.sample_id IS NULL THEN 'manual' END) AS locality_source,
        MAX(CASE WHEN mu.field = 'collection_year' AND mc.sample_id IS NULL THEN mu.value END) AS collection_year,
        MAX(CASE WHEN mu.field = 'collection_year' AND mc.sample_id IS NULL THEN 'manual' END) AS year_source,
        MAX(CASE WHEN mu.field = 'collection_date' AND mc.sample_id IS NULL THEN mu.value END) AS collection_date,
        MAX(CASE WHEN mu.field = 'phylotype' AND mc.sample_id IS NULL THEN mu.value END) AS phylotype,
        MAX(CASE WHEN mu.field = 'phylotype' AND mc.sample_id IS NULL THEN 'manual' END) AS phylotype_source,
        MAX(CASE WHEN mu.field = 'sequevar' AND mc.sample_id IS NULL THEN mu.value END) AS sequevar,
        MAX(CASE WHEN mu.field = 'sequevar' AND mc.sample_id IS NULL THEN 'manual' END) AS sequevar_source,
        MAX(CASE WHEN mu.field = 'sequence_type_ST' AND mc.sample_id IS NULL THEN mu.value END) AS sequence_type_ST,
        MAX(CASE WHEN mu.field = 'lineage' AND mc.sample_id IS NULL THEN mu.value END) AS lineage,
        MAX(CASE WHEN mu.field = 'race_or_pathotype' AND mc.sample_id IS NULL THEN mu.value END) AS race_or_pathotype
    FROM manual_unique mu
    LEFT JOIN manual_conflict mc ON mu.sample_id = mc.sample_id AND mu.field = mc.field
    GROUP BY mu.sample_id
)
SELECT
    s.sample_id,
    s.biosample_accession,
    s.strain,
    COALESCE(mp.organism_name, s.organism_name) AS organism_name,
    s.bioproject_accession,
    s.preferred_assembly,

    COALESCE(mp.host, sm.host_species) AS host,
    COALESCE(mp.cultivar, sm.host_cultivar) AS cultivar,
    COALESCE(mp.isolation_source, sm.isolation_source) AS isolation_source,

    COALESCE(mp.country, sm.country) AS country,
    COALESCE(mp.region, sm.admin1) AS region,
    COALESCE(mp.locality, sm.locality) AS locality,

    COALESCE(CAST(mp.collection_year AS INTEGER), sm.collection_year) AS collection_year,
    COALESCE(mp.collection_date, sm.collection_date_raw) AS collection_date,

    mp.phylotype,
    mp.sequevar,
    mp.sequence_type_ST,
    mp.lineage,
    mp.race_or_pathotype,

    gc.qc_class,

    COALESCE(mp.host_source, CASE WHEN sm.host_species IS NOT NULL THEN 'database' END) AS host_source,
    COALESCE(mp.cultivar_source, CASE WHEN sm.host_cultivar IS NOT NULL THEN 'database' END) AS cultivar_source,
    COALESCE(mp.isolation_source_source, CASE WHEN sm.isolation_source IS NOT NULL THEN 'database' END) AS isolation_source_source,
    COALESCE(mp.country_source, CASE WHEN sm.country IS NOT NULL THEN 'database' END) AS country_source,
    COALESCE(mp.region_source, CASE WHEN sm.admin1 IS NOT NULL THEN 'database' END) AS region_source,
    COALESCE(mp.locality_source, CASE WHEN sm.locality IS NOT NULL THEN 'database' END) AS locality_source,
    COALESCE(mp.year_source, CASE WHEN sm.collection_year IS NOT NULL THEN 'database' END) AS year_source,
    COALESCE(mp.phylotype_source, NULL) AS phylotype_source,
    COALESCE(mp.sequevar_source, NULL) AS sequevar_source,
    mp.host_manual_id

FROM samples s
LEFT JOIN sample_metadata sm ON s.sample_id = sm.sample_id
LEFT JOIN genome_qc_classification gc ON s.sample_id = gc.sample_id
LEFT JOIN manual_pivot mp ON s.sample_id = mp.sample_id
